***#Find total sales amount for each order.***

SELECT 

&#x20;   OrderID,

&#x20;   SUM(UnitPrice \* Quantity \* (1 - Discount)) AS TotalSales

FROM

&#x20;   Northwind.Order\_Details

GROUP BY OrderID;



***# Get top 5 customers based on total purchase amount.***



SELECT 

&#x20;   customers.CustomerID,

&#x20;   customers.CompanyName,

&#x20;   SUM(order\_details.UnitPrice \* order\_details.Quantity \* (1 - order\_details.Discount)) AS TotalSpent

FROM

&#x20;   Customers

&#x20;       JOIN

&#x20;   Orders ON customers.CustomerID = orders.CustomerID

&#x20;       JOIN

&#x20;   Order\_Details ON orders.OrderID = order\_details.OrderID

GROUP BY Customers.CustomerID , Customers.CompanyName

ORDER BY TotalSpent DESC

LIMIT 5;



***#Calculate total revenue generated by each product.***



SELECT 

&#x20;   products.ProductName,

&#x20;   SUM(order\_details.UnitPrice \* order\_details.Quantity \* (1 - order\_details.Discount)) AS Revenue

FROM

&#x20;   Products

&#x20;       JOIN

&#x20;   Order\_Details ON products.ProductID = order\_details.ProductID

GROUP BY products.ProductName

ORDER BY Revenue DESC;



***#Find monthly sales trend (year + month wise revenue).***



SELECT 

&#x20;   YEAR(orders.OrderDate) AS Year,

&#x20;   MONTH(orders.OrderDate) AS Month,

&#x20;   SUM(order\_details.UnitPrice \* order\_details.Quantity \* (1 - order\_details.Discount)) AS MonthlyRevenue

FROM

&#x20;   Orders

&#x20;       JOIN

&#x20;   Order\_Details ON order\_details.OrderID = orders.OrderID

GROUP BY Year , Month

ORDER BY Year , Month;

# **ADVANCED ANALYSIS**



***#Rank customers based on total spending*** 



SELECT 

&#x20;   CustomerID,

&#x20;   TotalSpent,

&#x20;   RANK() OVER (ORDER BY TotalSpent DESC) AS RankPosition

FROM (

&#x20;   SELECT 

&#x20;       customers.CustomerID,

&#x20;       SUM(order\_details.UnitPrice \* order\_details.Quantity \* (1 - order\_details.Discount)) AS TotalSpent

&#x20;   FROM Customers

&#x20;   JOIN Orders ON customers.CustomerID = orders.CustomerID

&#x20;   JOIN Order\_Details ON orders.OrderID = order\_details.OrderID

&#x20;   GROUP BY customers.CustomerID

) t;



***#Find the running total of sales over time (cumulative revenue).***



SELECT 

&#x20;   OrderDate,

&#x20;   DailyRevenue,

&#x20;   SUM(DailyRevenue) OVER (ORDER BY OrderDate) AS RunningTotal

FROM (

&#x20;   SELECT 

&#x20;       orders.OrderDate,

&#x20;       SUM(order\_details.UnitPrice \* order\_details.Quantity \* (1 - order\_details.Discount)) AS DailyRevenue

&#x20;   FROM Orders

&#x20;   JOIN Order\_Details ON orders.OrderID = order\_details.OrderID

&#x20;   GROUP BY orders.OrderDate

) t;



***#Identify repeat customers (customers with more than 1 order) and calculate their contribution to total revenue.***



WITH CustomerStats AS (

&#x20;   SELECT 

&#x20;       customers.CustomerID,

&#x20;       customers.CompanyName,

&#x20;       COUNT(DISTINCT orders.OrderID) AS OrderCount,

&#x20;       SUM(order\_details.UnitPrice \* order\_details.Quantity \* (1 - order\_details.Discount)) AS Revenue

&#x20;   FROM Customers

&#x20;   JOIN Orders ON customers.CustomerID = orders.CustomerID

&#x20;   JOIN Order\_Details ON orders.OrderID = order\_details.OrderID

&#x20;   GROUP BY customers.CustomerID, customers.CompanyName

)



SELECT 

&#x20;   CustomerID,

&#x20;   Revenue,

&#x20;   CASE 

&#x20;       WHEN OrderCount > 1 THEN 'Repeat Customer'

&#x20;       ELSE 'One-time Customer'

&#x20;   END AS CustomerType

FROM CustomerStats

ORDER BY Revenue DESC;





***#Calculate average order value per customer and classify them:***

***-- High Value***

***-- Medium Value***

***-- Low Value***



WITH OrderTotals AS (

&#x20;   SELECT 

&#x20;       orders.OrderID,

&#x20;       orders.CustomerID,

&#x20;       SUM(order\_details.UnitPrice \* order\_details.Quantity \* (1 - order\_details.Discount)) AS OrderTotal

&#x20;   FROM Orders 

&#x20;   JOIN Order\_Details ON orders.OrderID = order\_details.OrderID

&#x20;   GROUP BY order\_details.OrderID, orders.CustomerID

),

CustomerAOV AS (

&#x20;   SELECT 

&#x20;       CustomerID,

&#x20;       AVG(OrderTotal) AS AvgOrderValue

&#x20;   FROM OrderTotals

&#x20;   GROUP BY CustomerID

)



SELECT 

&#x20;   CustomerID,

&#x20;   AvgOrderValue,

&#x20;   CASE 

&#x20;       WHEN AvgOrderValue > 1000 THEN 'High Value'

&#x20;       WHEN AvgOrderValue > 500 THEN 'Medium Value'

&#x20;       ELSE 'Low Value'

&#x20;   END AS Segment

FROM CustomerAOV;





***#Find the most profitable shipping company based on total order value shipped.***



SELECT 

&#x20;   shippers.CompanyName AS Shipper,

&#x20;   SUM(order\_details.UnitPrice \* order\_details.Quantity \* (1 - order\_details.Discount)) AS TotalShippedValue

FROM

&#x20;   Shippers

&#x20;       JOIN

&#x20;   Orders ON shippers.ShipperID = orders.shipperID

&#x20;       JOIN

&#x20;   Order\_Details ON orders.OrderID = order\_details.OrderID

GROUP BY shippers.CompanyName

ORDER BY TotalShippedValue DESC;
